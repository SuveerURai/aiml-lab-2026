data= pd.read_csv("/home/sahyadri/AIML Dataset/train.csv")

]

# --------------------------------------------------
model= LinearRegression()

# Step8:Makepredictions

# Step9:Evaluatethemodel

mae= mean_absolute_error(y_test, y_pred)
